const express = require("express");
const { MongoClient } = require("mongodb");
const cors = require("cors");

const app = express();

app.use(cors());
app.use(express.json());
app.use(express.static("public"));

const url = "mongodb://localhost:27017";
const client = new MongoClient(url);

// Add Student - CREATE
app.post("/students", async (req, res) => {
    try {
        await client.connect();

        const db = client.db("StudentDB");
        const collection = db.collection("students");

        const existing = await collection.findOne({
            studentId: Number(req.body.studentId)
        });

        if (existing) {
            return res.status(400).send("Student ID already exists");
        }

        await collection.insertOne({
            studentId: Number(req.body.studentId),
            name: req.body.name,
            department: req.body.department
        });

        res.send("Student Added Successfully");
    } catch (error) {
        console.log(error);
        res.status(500).send("Error Adding Student");
    }
});

// View Students - READ
app.get("/students", async (req, res) => {
    try {
        await client.connect();

        const db = client.db("StudentDB");
        const collection = db.collection("students");

        const data = await collection.find().toArray();

        res.json(data);
    } catch (error) {
        console.log(error);
        res.status(500).send("Error Fetching Students");
    }
});

// Update Student - UPDATE
app.put("/students/:id", async (req, res) => {
    try {
        await client.connect();

        const db = client.db("StudentDB");
        const collection = db.collection("students");

        const result = await collection.updateOne(
            { studentId: Number(req.params.id) },
            {
                $set: {
                    name: req.body.name,
                    department: req.body.department
                }
            }
        );

        if (result.matchedCount === 0) {
            return res.status(404).send("Student Not Found");
        }

        res.send("Student Updated Successfully");
    } catch (error) {
        console.log(error);
        res.status(500).send("Error Updating Student");
    }
});

// Delete Student - DELETE
app.delete("/students/:id", async (req, res) => {
    try {
        await client.connect();

        const db = client.db("StudentDB");
        const collection = db.collection("students");

        const result = await collection.deleteOne({
            studentId: Number(req.params.id)
        });

        if (result.deletedCount === 0) {
            return res.status(404).send("Student Not Found");
        }

        res.send("Student Deleted Successfully");
    } catch (error) {
        console.log(error);
        res.status(500).send("Error Deleting Student");
    }
});

app.listen(3000, () => {
    console.log("Server Running on Port 3000");
});

<!DOCTYPE html>
<html ng-app="studentApp">

<head>

</head>

<body ng-controller="studentController">

    <h2>Student List</h2>

    <!-- Custom Directive -->
    <p student-module></p>

    <!-- Custom Filter Search -->
    <input type="text"
           ng-model="searchName"
           placeholder="Search by student name">

    <!-- Update Form -->
    <div ng-if="editing">

        <h3>Edit Student</h3>

        <input type="text"
               ng-model="editData.name"
               placeholder="Student Name"
               required>

        <br><br>

        <input type="text"
               ng-model="editData.department"
               placeholder="Department"
               required>

        <br><br>

        <button ng-click="updateStudent()">
            Update Student
        </button>

        <button ng-click="editing = false">
            Cancel
        </button>

    </div>

    <!-- Student Table -->
    <table>
        <tr>
            <th>ID</th>
            <th>Name</th>
            <th>Department</th>
            <th>Action</th>
        </tr>

        <tr ng-repeat="s in students | studentFilter:searchName">
            <td>{{s.studentId}}</td>
            <td>{{s.name}}</td>
            <td>{{s.department}}</td>

            <td>
                <button ng-click="editStudent(s)">
                    Edit
                </button>

                <button ng-click="deleteStudent(s.studentId)">
                    Delete
                </button>
            </td>
        </tr>

        <tr ng-if="students.length === 0">
            <td colspan="4">No students found</td>
        </tr>
    </table>

    <p>{{message}}</p>

    <a href="index.html">
        <button>Home</button>
    </a>

    <a href="add.html">
        <button>Add Student</button>
    </a>

    <script src="https://ajax.googleapis.com/ajax/libs/angularjs/1.8.2/angular.min.js"></script>

    <script>
        var app = angular.module("studentApp", []);

        // Controller
        app.controller("studentController",
        function($scope, $http) {

            $scope.students = [];
            $scope.editing = false;

            // Read Students
            function loadStudents() {
                $http.get("/students")
                .then(function(response) {
                    $scope.students = response.data;
                })
                .catch(function(error) {
                    $scope.message = "Error loading students";
                    console.log(error);
                });
            }

            loadStudents();

            // Select Student for Editing
            $scope.editStudent = function(s) {
                $scope.editData = angular.copy(s);
                $scope.editing = true;
                $scope.message = "";
            };

            // Update Student
            $scope.updateStudent = function() {

                $http.put(
                    "/students/" + $scope.editData.studentId,
                    $scope.editData
                )
                .then(function(response) {
                    $scope.message = response.data;
                    $scope.editing = false;
                    loadStudents();
                })
                .catch(function(error) {
                    $scope.message =
                        error.data || "Update Failed";
                    console.log(error);
                });

            };

            // Delete Student
            $scope.deleteStudent = function(id) {

                $http.delete("/students/" + id)
                .then(function(response) {
                    $scope.message = response.data;
                    loadStudents();
                })
                .catch(function(error) {
                    $scope.message =
                        error.data || "Delete Failed";
                });

            };

        });

        // Custom Filter
        app.filter("studentFilter", function() {

            return function(students, searchName) {

                if (!students) {
                    return [];
                }

                if (!searchName) {
                    return students;
                }

                var result = [];

                angular.forEach(students, function(s) {

                    if (s.name &&
                        s.name.toLowerCase().includes(
                            searchName.toLowerCase()
                        )) {
                        result.push(s);
                    }

                });

                return result;
            };

        });

        // Custom Directive: studentModule
        app.directive("studentModule", function() {

            return {
                restrict: "A",
                template: "<b>Student Management Records</b>"
            };

        });
    </script>

</body>
</html>
